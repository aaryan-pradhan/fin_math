\section{Martingale tests}

\begin{problem}[Doubled uniforms]
$U_1,U_2,\ldots$ independent, uniform on $(0,1]$. Is $X_0=1$, $X_n=2^nU_1\cdots U_n$ a martingale?
\uses{card:incr}{5}
\plug{$\eta_i=U_i$, $\E_P\eta=\frac12$, so $\prod_{i\le n}\eta_i/(\E_P\eta)^n=2^nU_1\cdots U_n$.}
\answer{Yes; $\E_PX_n=1$ for every $n$.}
\end{problem}

\begin{problem}[Exponential sums]
$\epsilon_i$ independent exponential with mean $1$, $S_n=\epsilon_1+\dots+\epsilon_n$. Is $X_n=2^ne^{-S_n}$ a martingale?
\uses{card:incr}{5}
\plug{$\eta_i=e^{-\epsilon_i}$, $\E_P\eta=\int_0^\infty e^{-u}e^{-u}du=\frac12$.}
\answer{Yes: $X_n=\prod\eta_i/(\frac12)^n$.}
\end{problem}

\begin{problem}[A martingale that goes to zero]
$\xi_i$ independent Bernoulli$(p)$. Is $X_n=p^{-n}\xi_1\cdots\xi_n$ a martingale, and what is $\lim X_n$?
\uses{card:incr}{5}
\plug{$\eta_i=\xi_i$, $\E_P\eta=p$. $X_n=p^{-n}$ w.p.\ $p^n$ and $0$ otherwise; once $0$, always $0$.}
\answer{A martingale with $\E_PX_n=1$, yet $X_n\to0$ with probability $1$ (Card~\ref{card:limits}, row 2: $P(X_n\ne0)=p^n\to0$).}
\end{problem}

\begin{problem}[The absorbed fair walk]
$X_{n+1}=X_n\pm1$ w.p.\ $\frac12$ if $X_n>0$, and $X_{n+1}=0$ if $X_n=0$; $X_0=i$. Is it a martingale? Probability of reaching $N\ge i$ before $0$?
\uses{card:onestep}{4}
\plug{$X_n>0$: $\frac12(X_n+1)+\frac12(X_n-1)=X_n$; $X_n=0$: stays $0$.}
\uses{card:ost}{1}
\plug{$M=X$, $x=i$, $a=0$, $b=N$: $P(\text{exit at }N)=\frac{i-0}{N-0}$.}
\answer{A non-negative martingale; $P=\frac iN$.}
\end{problem}

\begin{problem}[$S_n^2-n\sigma^2$ and the Poisson martingales]
(a) $Y_j$ i.i.d., mean $0$, variance $\sigma^2$: is $S_n^2-n\sigma^2$ a martingale? (b) $N_t$ with independent Poisson$(\lambda(t-s))$ increments: show $N_t-\lambda t$, $(N_t-\lambda t)^2-\lambda t$, $e^{uN_t-\lambda t(e^u-1)}$ are martingales; find $\E_P(N_{t+s}\given\F_t)$, $\E_P(N_{t+s}^2\given\F_t)$.
\uses{card:incr}{3}
\plug{(a) $\mu=0$: the second martingale $S_n^2-n\sigma^2$.}
\uses{card:incr}{7}
\plug{(b) read off the three; the martingale property of $N_t-\lambda t$ and $(N_t-\lambda t)^2-\lambda t$ gives the conditional moments.}
\answer{(a) Yes. (b) The three martingales; $\E_P(N_{t+s}\given\F_t)=N_t+\lambda s$, $\E_P(N_{t+s}^2\given\F_t)=(N_t+\lambda s)^2+\lambda s$.}
\end{problem}

\begin{problem}[Read the negation]
True or false: ``$|W_t|^2-t$ is \emph{not} a martingale''.
\uses{card:incr}{1}
\plug{$|W_t|^2=W_t^2$, and $W_t^2-t$ is the second martingale.}
\answer{False.}
\end{problem}

\section{Conditional expectations}

\begin{problem}[Conditioning a walk on its current value]
$X_i$ i.i.d., mean $\mu$, variance $\sigma^2$, $S_n=X_1+\dots+X_n$. Find $\E_P(S_{n+m}\given S_n)$, $\E_P(S_{n+m}^2\given S_n)$, $\E_P(X_1\given S_n)$, $\E_P(S_m\given S_n)$ ($m\le n$).
\uses{card:freeze}{4}
\plug{$K=S_n$, $Z=S_{n+m}-S_n$ with mean $m\mu$, variance $m\sigma^2$.}
\answer{$S_n+m\mu$ and $(S_n+m\mu)^2+m\sigma^2$.}
For the last two (not on a card): by symmetry each $X_k$, $k\le n$, has the same conditional mean given $S_n$, and these add to $S_n$.
\answer{$\E_P(X_1\given S_n)=\frac{S_n}n$; $\E_P(S_m\given S_n)=\frac mnS_n$.}
\end{problem}

\begin{problem}[Every integrable variable gives a martingale]
Show $X_t=\E_P(Y\given\F_t)$ is a martingale ($\E_P|Y|<\infty$).
\uses{card:freeze}{6}
\plug{tower with $s\le t$: $\E_P(X_t\given\F_s)=\E_P(Y\given\F_s)=X_s$.}
\answer{A martingale (adapted by definition, $\E_P|X_t|\le\E_P|Y|$).}
\end{problem}

\begin{problem}[Martingale increments are uncorrelated]
$X$ a square-integrable martingale, $t_1\le t_2\le t_3\le t_4$. Show $\E_P[(X_{t_2}-X_{t_1})(X_{t_4}-X_{t_3})]=0$.
\uses{card:freeze}{2}
\plug{$K=X_{t_2}-X_{t_1}$ is known at $t_3$, and $\E_P(X_{t_4}-X_{t_3}\given\F_{t_3})=0$ (\cref{card:onestep}, row 5); then average (row 6).}
\answer{$\E_P[K\cdot0]=0$.}
\end{problem}

\section{Stopping times and optional stopping}

\begin{problem}[Which random times are stopping times?]
$X$ positive and adapted. (i) $\tau=m$; (ii) $\tau_1\wedge\tau_2$; (iii) $\min\{t:X_{t+1}/X_t>1\}$; (iv) $\min\{t:\sum_{k\le t}X_k>X_t^2\}$; (v) $\max\{t\le T:X_t>10\}$.
\uses{card:st}{1}
\plug{(iv): the condition at time $t$ uses $X_0,\dots,X_t$.}
\uses{card:st}{2}
\plug{(i), (ii).}
\uses{card:st}{3}
\plug{(iii) needs $X_{t+1}$; (v) is a last exit.}
\answer{(i), (ii), (iv) are stopping times; (iii), (v) are not.}
\end{problem}

\begin{problem}[Win 10 or lose 5]
A fair $\pm1$ game from $0$ stops at $+10$ or $-5$. Probability of quitting a winner? Expected number of rounds?
\uses{card:ost}{1}
\plug{$x=0$, $a=-5$, $b=10$, $v=1$.}
\answer{$P=\frac{5}{15}=\frac13$; $\E_P\tau=\frac{5\cdot10}{1}=50$.}
\end{problem}

\begin{problem}[Two levels for Brownian motion]
$\tau$ = first time $W_t\in\{a,b\}$, $a<0<b$. Law of $W_\tau$ and $\E_P\tau$.
\uses{card:ost}{1}
\plug{$x=0$, $v=1$.}
\answer{$P(W_\tau=b)=\frac{-a}{b-a}$, $P(W_\tau=a)=\frac{b}{b-a}$; $\E_P\tau=-ab$.}
\end{problem}

\begin{problem}[The unfair walk between two barriers]
Steps $+1$ w.p.\ $p\ne\frac12$, $-1$ w.p.\ $q$; exit from $(a,b)$, $a<0<b$, start $0$. Law of $S_\tau$ and $\E_P\tau$.
\uses{card:ost}{4}
\plug{shift by $-a$: start $i=-a$, levels $0$ and $N=b-a$; $r=q/p$.}
\answer{$P(S_\tau=b)=\dfrac{1-r^{-a}}{1-r^{b-a}}$; $\E_P\tau=\dfrac{\E_PS_\tau}{p-q}=\dfrac{a+(b-a)P(S_\tau=b)}{p-q}$.}
\end{problem}

\begin{problem}[Up one, down two]
Steps $+1$ w.p.\ $\frac67$, $-2$ w.p.\ $\frac17$, $S_0=0$. (i) Is $2^{-S_n}$ a martingale? (ii) $\tau=\min\{n:2^{-S_n}\le0.1\}$; find $\E_P\tau$.
\uses{card:incr}{3}
\plug{(i) third martingale with $e^u=\frac12$: $\E_P2^{-Y}=\frac12\cdot\frac67+4\cdot\frac17=1$, so $2^{-S_n}$ itself.}
\uses{card:ost}{8}
\plug{(ii) $2^{-S_n}\le0.1\iff S_n\ge\log_210\approx3.32\iff S_n\ge4$; up-steps are $1$, so $S_\tau=4$; $m=\frac67-\frac27=\frac47$.}
\answer{(i) Yes. (ii) $\E_P\tau=\frac{4}{4/7}=7$.}
\end{problem}

\begin{problem}[Hitting a rising line from above]
$\tau=\min\{t:W_t=2t-4\}$. Find $\E_P\tau$, $\Var\tau$, $\E_PW_\tau$, $\E_PW_\tau^2$, $\E_Pe^{-s\tau}$.
\uses{card:ost}{5}
\plug{$c=-4$, $\beta=2$.}
\answer{$\E_P\tau=2$, $\Var\tau=\frac{4}{8}=0.5$; $\E_PW_\tau=0$, $\E_PW_\tau^2=\E_P\tau=2$ (first two identities).}
\uses{card:ost}{7}
\plug{$\frac12u^2-2u=s$, root $u=2-\sqrt{4+2s}$, $\E_Pe^{-s\tau}=e^{4u}$.}
\answer{$\E_Pe^{-s\tau}=e^{8-4\sqrt{2(s+2)}}$.}
\end{problem}

\begin{problem}[Parabolic boundaries]
$\tau=\min\{t:W_t=\pm\sqrt{a+bt}\}$, $a>0$, $0<b<1$. Find $\E_P\tau$.
\uses{card:ost}{6}
\answer{$\E_P\tau=\frac{a}{1-b}$.}
\end{problem}

\begin{problem}[Exit from $(-1,1)$: Laplace transform and variance]
$\tau$ = first time $|W_t|=1$. Find $\E_Pe^{-s\tau}$, $\E_P\tau$, $\Var\tau$.
\uses{card:ost}{7}
\plug{symmetric exit, $a=1$: $\E_Pe^{-s\tau}=1/\cosh\sqrt{2s}=1-s+\frac56s^2-\cdots$; match $1-s\E_P\tau+\frac{s^2}2\E_P\tau^2$.}
\answer{$\E_P\tau=1$, $\E_P\tau^2=\frac53$, $\Var\tau=\frac23$.}
\end{problem}

\begin{problem}[The doubling strategy]
Double the stake after each loss in a fair game; stop at the first win $\tau$. Law and mean of $\tau$; $Z_\tau$; does optional stopping hold?
\uses{card:stakes}{3}
\plug{$P(\tau=n)=2^{-n}$, $\E_P\tau=\sum n2^{-n}=2$.}
\answer{$Z_\tau=1$ always, so $\E_PZ_\tau=1\ne0$: optional stopping fails because the stakes (and losses before $\tau$) are unbounded.}
\end{problem}
