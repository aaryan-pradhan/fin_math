\chapter{Problems: Itô calculus and SDEs}
\label{ch:prob-ito}

\begin{question}
Can you compute a stochastic differential, test a martingale, and solve or verify an SDE in a few lines?
\end{question}

\begin{whatfor}
\begin{itemize}
  \item Distributions and moments of Itô integrals.
  \item Stochastic differentials by Itô's formula, and the zero-drift martingale test, including expectations obtained from it.
  \item Solving and verifying SDEs; writing Itô's formula as a short note.
\end{itemize}
\skipline{fs:prob-ito}
\end{whatfor}

\begin{prereq}
Chapters~16--18: Itô integral, isometry, quadratic variation, Itô process, Itô's formula, product rule, martingale test, OU and GBM SDEs. Chapter~7: optional stopping.
\end{prereq}

\section{Itô integrals}

\begin{problem}[Distributions of Itô integrals]
Find the distribution of $\int_0^1t\,dW_t$, the variance of $\int_0^1t^4\,dW_t$, and the distribution of $\int_0^ts\,dW_s$.
\begin{steps}
  \item Non-random integrands, so each integral is $N\big(0,\int f^2\big)$.
  \item $\int_0^1t^2dt=\frac13$: $N(0,\frac13)$. $\int_0^1t^8dt=\frac19\approx0.11$. $\int_0^ts^2ds=\frac{t^3}3$: $N(0,\frac{t^3}3)$.
\end{steps}
\end{problem}

\begin{problem}[Isometry with two integrands]
For adapted $f,g$ with finite $\int\E_Pf^2$, $\int\E_Pg^2$, show $\E_P\big[\int_0^tf\,dW\int_0^tg\,dW\big]=\int_0^t\E_P(f_sg_s)\,ds$.
\begin{steps}
  \item Apply the isometry to $f+g$ and to $f$, $g$ separately: $\E_P\big(\int(f+g)dW\big)^2=\int\E_P(f+g)^2$.
  \item Expand both sides: the squares match by the isometry, so the cross terms match: $2\E_P\big[\int f\,dW\int g\,dW\big]=2\int\E_P(fg)$.
\end{steps}
\end{problem}

\section{Stochastic differentials and the martingale test}

\begin{problem}[Three differentials]
Compute $d\cos W_t$, $d(e^tW_t)$, and $d\,e^{-2X_t}$ where $X_t=t+W_t$. Which of $e^tW_t$ and $e^{-2X_t}$ are martingales?
\begin{steps}
  \item $f=\cos x$: $d\cos W_t=-\sin W_t\,dW_t-\frac12\cos W_t\,dt$.
  \item $f=e^tx$ ($\partial_{xx}f=0$): $d(e^tW_t)=e^tW_t\,dt+e^t\,dW_t$. The $dt$-term is not $0$: \textbf{not} a martingale. (Directly: $\E_P(e^tW_t\given\F_s)=e^tW_s\ne e^sW_s$.)
  \item $f=e^{-2x}$, $dX=dt+dW$, $(dX)^2=dt$: $d\,e^{-2X_t}=-2e^{-2X_t}(dt+dW_t)+\frac12\cdot4e^{-2X_t}dt=-2e^{-2X_t}\,dW_t$. No $dt$-term: a martingale.
\end{steps}
\end{problem}

\begin{problem}[The exponential martingale]
Let $X_t=e^{W_t-t/2}$. Is it a martingale? Compute $dX_t$.
\begin{steps}
  \item $f(t,x)=e^{x-t/2}$: $\partial_tf=-\frac12f$, $\partial_xf=\partial_{xx}f=f$.
  \item $dX_t=\big(-\frac12X_t+\frac12X_t\big)dt+X_t\,dW_t=X_t\,dW_t$: no drift, so yes (as in Chapter~6).
\end{steps}
More generally, for non-random $g$, $Y_t=\exp\big\{\int_0^tg\,dW-\frac12\int_0^tg^2ds\big\}$ has $dY_t=Y_tg_t\,dW_t$ and is a martingale.
\end{problem}

\begin{problem}[An expectation by the martingale trick]
Let $X_t=e^{t/2}\cos W_t$. Compute $dX_t$ and then $\E_P(\cos W_t)$.
\begin{steps}
  \item $f=e^{t/2}\cos x$: $\partial_tf=\frac12f$, $\partial_xf=-e^{t/2}\sin x$, $\partial_{xx}f=-f$.
  \item $dX_t=\frac12X_t\,dt-e^{t/2}\sin W_t\,dW_t-\frac12X_t\,dt=-e^{t/2}\sin W_t\,dW_t$.
  \item No drift, so $X$ is a martingale and $\E_PX_t=X_0=1$: $e^{t/2}\E_P\cos W_t=1$, i.e.\ $\E_P(\cos W_t)=e^{-t/2}$.
\end{steps}
Never compute $\int\cos x\,e^{-x^2/2t}dx$ by hand: the zero drift does it for you. Likewise $e^{t/2}\sin W_t$ is a martingale, so $\E_P\sin W_t=0$.
\end{problem}

\begin{problem}[Which are martingales?]
Decide for $e^{t/2}\sin W_t$, $e^t\cos W_t$, $e^{t}\sin W_t$, $e^{t/2}\cos W_t$.
\begin{steps}
  \item For $g=e^{ct}\sin x$ or $e^{ct}\cos x$: $\partial_tg+\frac12\partial_{xx}g=(c-\frac12)g$.
  \item Zero exactly when $c=\frac12$: the first and last are martingales, the middle two are not.
\end{steps}
\end{problem}

\begin{problem}[Hitting a line, once more]
$T=\min\{t:W_t=2-3t\}$. Find $\E_P(T)$ by optional stopping.
\begin{steps}
  \item $W_T=2-3T$ and $\E_PW_T=0$.
  \item $2-3\E_PT=0$, so $\E_P(T)=\frac23\approx0.67$.
\end{steps}
\end{problem}

\section{Solving and verifying SDEs}

\begin{problem}[An OU process as a stochastic integral]
Solve $dX_t=-X_t\,dt+dW_t$, $X_0=x_0$. Is it an OU process? Find $\E_PX_t$ and $\Var X_t$.
\begin{steps}
  \item Integrating factor $e^t$: $d(e^tX_t)=e^tX_t\,dt+e^t\,dX_t=e^t\,dW_t$.
  \item Integrate: $X_t=x_0e^{-t}+\int_0^te^{-(t-s)}dW_s$. Yes: OU with $r=1$, $\sigma=1$.
  \item $\E_PX_t=x_0e^{-t}$; $\Var X_t=\int_0^te^{-2(t-s)}ds=\frac12(1-e^{-2t})\to\frac12$.
\end{steps}
\end{problem}

\begin{problem}[A square-root SDE]
Show that $X_t=W_t^2$ solves $dX_t=dt+2\sqrt{X_t}\,dW_t$, $X_0=0$, and find $\E_PX_t$ and $\Var X_t$.
\begin{steps}
  \item Itô with $f=x^2$: $d(W_t^2)=2W_t\,dW_t+dt$, and $X_0=0$.
  \item $2W_t=2\sqrt{X_t}$ when $W_t\ge0$. (Strictly $2\sqrt{X_t}=2|W_t|$; the process $\int\text{sign}(W)\,dW$ is again a Brownian motion, so $X$ solves the equation driven by that Brownian motion. For an exam, the first step is the expected answer.)
  \item $\E_PX_t=\E_PW_t^2=t$; $\Var X_t=\E_PW_t^4-t^2=3t^2-t^2=2t^2$.
\end{steps}
\end{problem}

\begin{problem}[Verifying a guessed GBM]
$dS_t=0.2S_t\,dt+S_t\,dW_t$, $S_0=5$. Find $a,b,c$ with $S_t=ce^{at+bW_t}$, and the SDE of $X_t=1/S_t$.
\begin{steps}
  \item Itô on $ce^{at+bx}$: drift $(a+\frac12b^2)S$, noise $bS$. Match: $b=1$, $a=-0.3$, $c=5$.
  \item $X_t=0.2e^{0.3t-W_t}$: $dX_t=(0.3+\frac12)X_t\,dt-X_t\,dW_t=0.8X_t\,dt-X_t\,dW_t$, $X_0=0.2$.
\end{steps}
\end{problem}

\begin{problem}[Short note: derive Itô's formula]
Write a short derivation of Itô's formula for $Y_t=f(t,X_t)$, $dX_t=a\,dt+b\,dW_t$.
\begin{steps}
  \item Taylor to second order: $dY=\partial_tf\,dt+\partial_xf\,dX+\frac12\partial_{tt}f\,dt^2+\partial_{tx}f\,dt\,dX+\frac12\partial_{xx}f\,(dX)^2$.
  \item The table $dt^2=dt\,dW=0$, $dW^2=dt$ kills $dt^2$ and $dt\,dX$, and gives $(dX)^2=b^2dt$; this is because $\sum(\Delta W)^2\to t$.
  \item Result: $dY=\big(\partial_tf+a\,\partial_xf+\frac12b^2\partial_{xx}f\big)dt+b\,\partial_xf\,dW$. Example: $f=x^2$, $X=W$ gives $d(W^2)=dt+2W\,dW$.
\end{steps}
\end{problem}

\formulasheet{fs:prob-ito}

\begin{card}{1. Expectation by zero drift}
\Formula
If $g(t,W_t)$ has $\partial_tg+\frac12\partial_{xx}g=0$, then $\E_Pg(t,W_t)=g(0,0)$.
\UseWhen
``Using part (i) or otherwise, evaluate $\E_P(\cos W_t)$'', expectations of trigonometric or polynomial functions of $W$.
\HowTo
\begin{enumerate}
  \item Find a time factor $h(t)$ that kills the drift of $h(t)\phi(W_t)$.
  \item Then $h(t)\E_P\phi(W_t)=h(0)\phi(0)$.
\end{enumerate}
\Caveats
\cav{Check the drift is exactly $0$ for all $x$, not only at $x=0$.}
\begin{cardex}
$\E_P\cos W_t=e^{-t/2}$; $\E_P(W_t^3-3tW_t)=0$; $\E_P\,e^{W_t}=e^{t/2}$.
\end{cardex}
\end{card}

\begin{card}{2. Integrating factor for linear SDEs}
\Formula
$dX=-rX\,dt+\sigma dW$: $d(e^{rt}X)=\sigma e^{rt}dW$; $X_t=x_0e^{-rt}+\sigma\int_0^te^{-r(t-s)}dW_s$.
\UseWhen
``Solve the SDE as a stochastic integral''.
\HowTo
\begin{enumerate}
  \item Multiply by $e^{rt}$ (product rule, no correction).
  \item Integrate; mean from the first term, variance by the isometry.
\end{enumerate}
\Caveats
\cav{The integrating factor removes $X$ only when the noise does not contain $X$.}
\begin{cardex}
$dX=-X\,dt+dW$: $\Var X_t=\frac12(1-e^{-2t})$.
\end{cardex}
\end{card}
